Eduwide - Flutter MVC Architecture
===================================
lib/
├── main.dart                      # App entry point, timezone & Firebase init
├── auth_service.dart              # Re-export bridge to services/auth_service.dart
│
├── core/                          # Global App Configuration & Design Tokens
│   ├── app_colors.dart            # Primary palette, backgrounds, status colors
│   ├── app_config.dart            # Cloudinary settings, collection constants, default images
│   ├── dashboard_wrapper.dart     # Role-based router (Admin/Teacher/Student)
│   └── firebase_options.dart      # Platform Firebase options
│
├── models/                        # (M in MVC) Data Models & Serialization
│   ├── user_model.dart            # AppUser / UserModel with role & lockout helpers
│   ├── class_model.dart           # TeacherClassModel for class management & fees
│   ├── meeting_model.dart         # MeetingModel for Zoom meeting management
│   └── enrollment_model.dart      # EnrollmentModel for student-class enrollments
│
├── controllers/                   # (C in MVC) Business Logic & State Operations
│   ├── auth_controller.dart       # Sign in, lockout management, sign up, password reset
│   ├── user_controller.dart       # User CRUD, role updates, account approvals/locking
│   ├── class_controller.dart      # Class listings, fee settings, enrollments, payment approvals
│   └── meeting_controller.dart    # Zoom meetings CRUD, join tracking, status filters
│
├── services/                      # External APIs & Infrastructure
│   ├── auth_service.dart          # FirebaseAuth service integration
│   └── cloudinary_service.dart    # Cloudinary multipart upload service
│
└── views/                         # (V in MVC) UI Layer (Screens, Dialogs, Drawers)
    ├── auth/                      # Authentication Views
    │   ├── start_page.dart
    │   ├── login_screen.dart
    │   ├── signup_screen.dart
    │   ├── forgot_password_screen.dart
    │   ├── firebase_load_check_screen.dart
    │   └── simple_preload_screen.dart
    ├── dashboard/                 # Role-based Dashboard View
    │   └── dashboard_wrapper.dart
    ├── admin/                     # Administrator Views
    │   ├── admin_dashboard.dart
    │   ├── accounts_management_screen.dart
    │   ├── class_management_screen.dart
    │   ├── class_enrollment_screen.dart
    │   ├── admin_all_meeting.dart
    │   ├── admin_resource_upload.dart
    │   ├── admin_profile.dart
    │   └── admin_drawer.dart
    ├── teacher/                   # Teacher Views
    │   ├── teacher_dashboard.dart
    │   ├── teacher_class_dashbord.dart
    │   ├── zoom_meeting_screen.dart
    │   ├── teacher_profile.dart
    │   └── teacher_drawer.dart
    └── student/                   # Student Views
        ├── user_dashboard.dart
        ├── user_class_detail_screen.dart
        ├── user_profile.dart
        └── student_drawer.dart
